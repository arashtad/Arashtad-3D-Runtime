Pass a partial name to playAnimation(). The runtime first checks for an exact name and then searches for a matching animation name. 

<arashtad-model
    id="model_1"
    src="../../../assets/3d/linux-animated.glb"
    position="new BABYLON.Vector3(0, -0.5, 0)"
    scaling="new BABYLON.Vector3(5, 5, 5)"
></arashtad-model>

<script type="module">

    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime-1.1.0.min.js';

    const canvas = document.getElementById('animation-partial');
    const scene = await ready(canvas);

    const modelRoot =
        scene.getTransformNodeByName('model_1-root');

    modelRoot.playAnimation('hap');

</script>
