Each model receives its own TransformNode animation API after loading. 

<arashtad-model
    id="model_1"
    src="../../../assets/3d/linux-animated.glb"
    position="new BABYLON.Vector3(-2, -0.5, 0)"
    scaling="new BABYLON.Vector3(4, 4, 4)"
></arashtad-model>

<arashtad-model
    id="model_2"
    src="../../../assets/3d/linux-animated.glb"
    position="new BABYLON.Vector3(2, -0.5, 0)"
    scaling="new BABYLON.Vector3(4, 4, 4)"
></arashtad-model>

<script type="module">

    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime-1.1.0.min.js';

    const canvas = document.getElementById('multiple-animated');
    const scene = await ready(canvas);

    const modelRoot1 =
        scene.getTransformNodeByName('model_1-root');

    const modelRoot2 =
        scene.getTransformNodeByName('model_2-root');

    modelRoot1.playAnimation('happy');
    modelRoot2.playAnimation('foot');

</script>
