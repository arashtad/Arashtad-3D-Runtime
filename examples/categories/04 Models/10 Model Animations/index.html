Model animation methods are available through the runtime model target and can be executed with runtime actions. 

<arashtad-model
    id="model_1"
    src="../../../assets/3d/linux-animated.glb"
>
    <action execute="console.log(target.getAnimations())" />
    <action execute="target.playAnimation(0)" />
    <action execute="target.playAnimation('happy')" />
    <action execute="target.pauseAnimation()" />
    <action execute="target.pauseAnimation(0)" />
    <action execute="target.resumeAnimation()" />
    <action execute="target.resumeAnimation(0)" />
    <action execute="target.stopAnimation()" />
    <action execute="target.stopAnimation(0)" />
    <action execute="target.setAnimationSpeed(20)" />
    <action execute="target.setAnimationSpeed(0.5, 0)" />
</arashtad-model>
