Basic Mesh

56

Create a basic Babylon.js mesh directly from the runtime using the declarative mesh element.

Mesh Creation

The arashtad-mesh element creates a mesh through Babylon.js MeshBuilder.

Box Mesh

This example creates a box with a size of 2 at the scene origin.

Source

Define the mesh type and its Babylon.js MeshBuilder options directly as HTML attributes. 

<canvas id="basic-mesh">

    <arashtad-runtime resize="engine.resize()"></arashtad-runtime>

    <arashtad-camera type="ArcRotateCamera" id="camera"
        args="Math.PI / 2, Math.PI / 2.4, 6, new BABYLON.Vector3(0, 0, 0)"
        minZ="0.1"
        maxZ="1000"></arashtad-camera>

    <arashtad-light type="HemisphericLight" id="light"
        args="new BABYLON.Vector3(0, 1, 0)"
        intensity="1.3"></arashtad-light>

    <arashtad-mesh type="Box" id="box"
        size="2"
        position="new BABYLON.Vector3(0, 0, 0)">
    </arashtad-mesh>

</canvas>

<script type="module">

    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime-1.1.0.min.js';

    const canvas = document.getElementById('basic-mesh');
    const scene = await ready(canvas);

</script>
