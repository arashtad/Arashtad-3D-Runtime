Mesh Enabled State

77

Enable or disable a mesh at runtime using its Babylon.js enabled state.

Enabled

Enable or Disable the Mesh

Toggle the mesh's enabled state without changing its visibility property.

Mesh State

Source

The switch changes the mesh's enabled state through the Babylon.js scene. 

<arashtad-mesh
    type="Box"
    id="enabled-mesh"
    size="2"
    position="new BABYLON.Vector3(0, 0, 0)">
</arashtad-mesh>

<script type="module">
    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime.1.0.0.min.js';

    const canvas = document.getElementById('mesh-enabled');

    const scene = await ready(canvas);

    const mesh = scene.meshes.find(
        mesh => mesh.name === 'enabled-mesh'
    );

    const toggle = document.getElementById('enabled-toggle');
    const label = document.getElementById('enabled-label');
    const status = document.getElementById('enabled-status');

    toggle.addEventListener('change', () => {
        mesh.setEnabled(toggle.checked);

        label.textContent = toggle.checked ? 'Enabled' : 'Disabled';
        status.innerHTML = `<strong>${toggle.checked ? 'Enabled' : 'Disabled'}</strong>`;
    });
</script>
