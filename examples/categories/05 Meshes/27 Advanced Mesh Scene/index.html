Use mesh triggers to control materials, mesh transforms, lighting, and camera properties directly from an interactive runtime scene.

Box · Sphere · Cylinder · Ground

Interactive Mesh Controls

Each mesh uses a native runtime trigger to modify another part of the scene or its own properties.

Interactions

Box - Changes its material color randomly.

Sphere - Toggles its scale.

Cylinder - Changes directional light color and intensity.

Ground - Changes the camera angle and radius.

Source

Mesh events can directly access registered runtime objects and Babylon.js properties. 

<arashtad-mesh
    type="Box"
    id="box"
    size="2"
    position="new BABYLON.Vector3(-3, 1, 0)"
    material="boxMaterial"
    isPickable="true"
    OnPickTrigger="
        box.material.diffuseColor =
            new BABYLON.Color3(
                Math.random(),
                Math.random(),
                Math.random()
            );
    ">
</arashtad-mesh>

<arashtad-mesh
    type="Sphere"
    id="sphere"
    diameter="2"
    segments="32"
    position="new BABYLON.Vector3(0, 1, 0)"
    material="sphereMaterial"
    isPickable="true"
    OnPickTrigger="
        sphere.scaling.x =
            sphere.scaling.x === 1 ? 1.5 : 1;

        sphere.scaling.y =
            sphere.scaling.y === 1 ? 1.5 : 1;

        sphere.scaling.z =
            sphere.scaling.z === 1 ? 1.5 : 1;
    ">
</arashtad-mesh>

<arashtad-mesh
    type="Cylinder"
    id="cylinder"
    height="2.5"
    diameter="1.8"
    tessellation="32"
    position="new BABYLON.Vector3(3, 1.25, 0)"
    material="cylinderMaterial"
    isPickable="true"
    OnPickTrigger="
        directionalLight.diffuse =
            directionalLight.diffuse.equals(
                new BABYLON.Color3(1, 0.2, 0.1)
            )
            ? new BABYLON.Color3(1, 1, 1)
            : new BABYLON.Color3(1, 0.2, 0.1);

        directionalLight.intensity =
            directionalLight.intensity === 2
                ? 0.5
                : 2;
    ">
</arashtad-mesh>

<arashtad-mesh
    type="Ground"
    id="ground"
    width="14"
    height="8"
    position="new BABYLON.Vector3(0, 0, 0)"
    material="groundMaterial"
    isPickable="true"
    OnPickTrigger="
        camera.alpha =
            camera.alpha === 1.57
                ? 2.2
                : 1.57;

        camera.beta = 1.15;

        camera.radius =
            camera.radius === 12
                ? 9
                : 12;
    ">
</arashtad-mesh>
