Multiple Lights

Combine multiple colored lights with different directions, intensities, and color temperatures to create a dark, cinematic, mysterious 3D environment.

Light Multiple Lights Cinematic

What this demonstrates

Multiple declarative lights can be combined to build complex lighting environments without creating the lights manually in JavaScript.

Ambient Light

A low-intensity HemisphericLight provides subtle ambient illumination while preserving the dark atmosphere.

Cinematic Colors

Warm orange, cool blue, green, and purple DirectionalLight sources create contrasting areas of illumination around the model.

Dark Environment

A nearly black ground material and very low ambient contribution keep the scene dark while allowing the colored lights to define the model.

Linux Model

The Linux model is loaded declaratively with arashtad-model and positioned close to the camera for a stronger presentation.

Source

Multiple lights, a model, and materials are defined directly inside the canvas. 

<canvas id="multiple-lights">

    <arashtad-runtime resize="engine.resize()">
    </arashtad-runtime>

    <arashtad-camera
        type="ArcRotateCamera"
        id="camera"
        args="1.57, 1.15, 9, new BABYLON.Vector3(0, -2, -5)"
        lowerRadiusLimit="6"
        upperRadiusLimit="14"
        lowerBetaLimit="0.7"
        upperBetaLimit="1.45"
        wheelPrecision="60"
        inertia="0.8">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="ambient"
        args="new BABYLON.Vector3(0, 1, 0)"
        intensity="0.18"
        diffuse="new BABYLON.Color3(0.12, 0.2, 0.3)"
        groundColor="new BABYLON.Color3(0.015, 0.02, 0.025)"
        specular="new BABYLON.Color3(0.08, 0.08, 0.1)">
    </arashtad-light>

    <arashtad-light
        type="DirectionalLight"
        id="key"
        args="new BABYLON.Vector3(-0.45, -0.8, 0.35)"
        position="new BABYLON.Vector3(5, 7, -6)"
        intensity="2.1"
        diffuse="new BABYLON.Color3(1, 0.3, 0.08)"
        specular="new BABYLON.Color3(0.35, 0.15, 0.05)">
    </arashtad-light>

    <arashtad-light
        type="DirectionalLight"
        id="rim"
        args="new BABYLON.Vector3(0.55, -0.7, -0.45)"
        position="new BABYLON.Vector3(-6, 6, 5)"
        intensity="2.6"
        diffuse="new BABYLON.Color3(0.08, 0.3, 1)"
        specular="new BABYLON.Color3(0.08, 0.15, 0.4)">
    </arashtad-light>

    <arashtad-light
        type="DirectionalLight"
        id="greenFill"
        args="new BABYLON.Vector3(0.65, -0.55, 0.6)"
        position="new BABYLON.Vector3(-7, 4, -4)"
        intensity="1.4"
        diffuse="new BABYLON.Color3(0.04, 0.65, 0.35)"
        specular="new BABYLON.Color3(0.05, 0.2, 0.12)">
    </arashtad-light>

    <arashtad-light
        type="DirectionalLight"
        id="purpleBack"
        args="new BABYLON.Vector3(-0.35, -0.65, -0.7)"
        position="new BABYLON.Vector3(4, 5, 8)"
        intensity="1.8"
        diffuse="new BABYLON.Color3(0.42, 0.08, 0.75)"
        specular="new BABYLON.Color3(0.18, 0.05, 0.3)">
    </arashtad-light>

    <arashtad-model
        id="linux"
        src="../../../assets/3d/linux.glb"
        position="new BABYLON.Vector3(0, 0, 0)"
        scaling="new BABYLON.Vector3(1.15, 1.15, 1.15)">
    </arashtad-model>

    <arashtad-material
        type="StandardMaterial"
        id="groundMaterial"
        diffuseColor="new BABYLON.Color3(0.008, 0.012, 0.016)"
        specularColor="new BABYLON.Color3(0.015, 0.02, 0.025)"
        roughness="0.92">
    </arashtad-material>

    <arashtad-material
        type="StandardMaterial"
        id="accentMaterial"
        diffuseColor="new BABYLON.Color3(0.015, 0.07, 0.055)"
        emissiveColor="new BABYLON.Color3(0.005, 0.025, 0.02)"
        specularColor="new BABYLON.Color3(0.04, 0.08, 0.07)">
    </arashtad-material>

    <arashtad-mesh
        type="Box"
        id="platform"
        width="4.8"
        height="0.18"
        depth="4.8"
        position="new BABYLON.Vector3(0, -1.02, 0)"
        material="groundMaterial">
    </arashtad-mesh>

    <arashtad-mesh
        type="Cylinder"
        id="accent-platform"
        diameter="3.8"
