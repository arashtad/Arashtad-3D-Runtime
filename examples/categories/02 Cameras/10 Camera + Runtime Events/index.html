Camera + Runtime Events

Control an ArcRotateCamera directly from interactive scene objects using Babylon.js native action triggers.

Camera Native Triggers OnPickTrigger

What this demonstrates

Babylon.js native object triggers can directly control the registered camera without using runtime action tags.

Native Pick Triggers

Each interactive mesh uses OnPickTrigger directly on the runtime mesh element.

Camera Orbit

The orange and blue controls change the camera's alpha, rotating the view around the scene.

Camera Zoom

The green and red controls modify the camera's radius, creating zoom-in and zoom-out controls directly inside the scene.

Source

Babylon.js native ActionManager triggers are declared directly on the interactive meshes. 

<canvas
    id="camera-runtime-events"
    clearColor="new BABYLON.Color4(0.04, 0.055, 0.05, 1)">

    <arashtad-runtime resize="engine.resize()">
    </arashtad-runtime>

    <arashtad-camera
        type="ArcRotateCamera"
        id="camera"
        args="BABYLON.Tools.ToRadians(45), BABYLON.Tools.ToRadians(65), 12, new BABYLON.Vector3(0, 0, 0)"
        lowerBetaLimit="BABYLON.Tools.ToRadians(20)"
        upperBetaLimit="BABYLON.Tools.ToRadians(85)"
        lowerRadiusLimit="5"
        upperRadiusLimit="18"
        inertia="0.8">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="light"
        args="new BABYLON.Vector3(0, 1, 0)"
        intensity="1.2">
    </arashtad-light>

    <arashtad-light
        type="DirectionalLight"
        id="sun"
        args="new BABYLON.Vector3(-1, -2, -1)"
        intensity="1.5"
        position="new BABYLON.Vector3(5, 8, 5)">
    </arashtad-light>

    <arashtad-material
        type="StandardMaterial"
        id="center-material"
        diffuseColor="new BABYLON.Color3(0.15, 0.64, 0.55)"
        specularColor="new BABYLON.Color3(0.2, 0.2, 0.2)">
    </arashtad-material>

    <arashtad-material
        type="StandardMaterial"
        id="left-material"
        diffuseColor="new BABYLON.Color3(1, 0.45, 0.05)"
        specularColor="new BABYLON.Color3(0.2, 0.2, 0.2)">
    </arashtad-material>

    <arashtad-material
        type="StandardMaterial"
        id="right-material"
        diffuseColor="new BABYLON.Color3(0.2, 0.45, 1)"
        specularColor="new BABYLON.Color3(0.2, 0.2, 0.2)">
    </arashtad-material>

    <arashtad-material
        type="StandardMaterial"
        id="zoom-in-material"
        diffuseColor="new BABYLON.Color3(0.25, 0.8, 0.35)"
        specularColor="new BABYLON.Color3(0.2, 0.2, 0.2)">
    </arashtad-material>

    <arashtad-material
        type="StandardMaterial"
        id="zoom-out-material"
        diffuseColor="new BABYLON.Color3(0.8, 0.25, 0.3)"
        specularColor="new BABYLON.Color3(0.2, 0.2, 0.2)">
    </arashtad-material>

    <arashtad-mesh
        type="Box"
        id="center"
        size="2"
        material="center-material"
        position="new BABYLON.Vector3(0, 0, 0)"
        OnPickTrigger="camera.alpha = BABYLON.Tools.ToRadians(45); camera.radius = 12;">
    </arashtad-mesh>

    <arashtad-mesh
        type="Box"
        id="left"
        size="1.5"
        material="left-material"
        position="new BABYLON.Vector3(-4, 0, 0)"
        OnPickTrigger="camera.alpha -= BABYLON.Tools.ToRadians(45);">
    </arashtad-mesh>

    <arashtad-mesh
        type="Box"
        id="right"
        size="1.5"
        material="right-material"
        position="new BABYLON.Vector3(4, 0, 0)"
        OnPickTrigger="camera.alpha += BABYLON.Tools.ToRadians(45);">
    </arashtad-mesh>

    <arashtad-mesh
        type="Box"
        id="front"
        size="1.5"
        material="zoom-in-material"
        position="new BABYLON.Vector3(0, 0, 4)"
        OnPickTrigger="camera.radius = Math.max(5, camera.radius - 2);">
    </arashtad-mesh>

    <arashtad-mesh
        type="Box"
        id="back"
        size="1.5"
        material="zoom-out-material"
        position="new BABYLON.Vector3(0, 0, -4)"
        OnPickTrigger="camera.radius = Math.min(18, camera.radius + 2);">
    </arashtad-mesh>

    <arashtad-mesh
        type="Ground"
        id="ground"
        width="14"
        height="14"
        position="new BABYLON.Vector3(0, -1, 0)">
    </arashtad-mesh>

</canvas>
