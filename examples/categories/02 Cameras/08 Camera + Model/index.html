Camera + Model

Load a 3D model declaratively and position an ArcRotateCamera around it for an interactive model-viewing scene.

Camera Model ArcRotateCamera

What this demonstrates

Combining a declarative camera with a runtime-loaded 3D model.

Camera

The ArcRotateCamera orbits around the model target and allows the viewer to rotate and zoom the scene.

Model

The arashtad-model tag loads the GLB model through Babylon.js SceneLoader.

Source

A camera, light, model, and ground are declared directly inside the runtime canvas. 

<canvas id="camera-model">

    <arashtad-runtime
        resize="engine.resize()">
    </arashtad-runtime>

    <arashtad-camera
        type="ArcRotateCamera"
        id="camera"
        args="BABYLON.Tools.ToRadians(45), BABYLON.Tools.ToRadians(70), 6, new BABYLON.Vector3(0, 0.5, 0)"
        lowerRadiusLimit="3"
        upperRadiusLimit="12">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="light"
        args="new BABYLON.Vector3(0, 1, 0)"
        intensity="1.2">
    </arashtad-light>

    <arashtad-model
        id="model"
        src="../../../assets/3d/linux.glb">
    </arashtad-model>

    <arashtad-mesh
        type="Ground"
        id="ground"
        width="12"
        height="12">
    </arashtad-mesh>

</canvas>

<script src="../../../../lib/babylon/babylon.js"></script>

<script type="module">
    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime.1.0.0.min.js';

    const canvas = document.getElementById('camera-model');

    await ready(canvas);
</script>
