UniversalCamera - Full Configuration

Configure a UniversalCamera with its position, target, movement speed, angular sensitivity, inertia, collision settings, gravity, and camera clipping properties.

Camera UniversalCamera Full Configuration

What this demonstrates

A UniversalCamera configured through declarative Babylon.js properties.

Movement

  • speed
  • angularSensibility
  • inertia

Collision & Gravity

  • checkCollisions
  • applyGravity
  • ellipsoid
  • ellipsoidOffset

Clipping

  • minZ
  • maxZ
  • Camera target
  • Camera position

Source

UniversalCamera properties are declared directly on the runtime camera element. 

<canvas id="universal-camera-full">

    <arashtad-runtime
        resize="engine.resize()">
    </arashtad-runtime>

    <arashtad-camera
        type="UniversalCamera"
        id="camera"
        args="new BABYLON.Vector3(0, 2, -10)"
        setTarget="new BABYLON.Vector3(0, 1, 0)"
        speed="0.25"
        angularSensibility="500"
        inertia="0.8"
        minZ="0.1"
        maxZ="1000"
        checkCollisions="true"
        applyGravity="true"
        ellipsoid="new BABYLON.Vector3(0.5, 1, 0.5)"
        ellipsoidOffset="new BABYLON.Vector3(0, 0.5, 0)">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="light"
        args="new BABYLON.Vector3(0, 1, 0)"
        intensity="1.2">
    </arashtad-light>

    <arashtad-mesh
        type="Ground"
        id="ground"
        width="20"
        height="20"
        position="new BABYLON.Vector3(0, -1, 0)">
    </arashtad-mesh>

    <arashtad-mesh
        type="Box"
        id="box"
        size="2"
        position="new BABYLON.Vector3(0, 1, 0)">
    </arashtad-mesh>

</canvas>

<script src="../../../../lib/babylon/babylon.js"></script>

<script type="module">
    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime.1.0.0.min.js';

    const canvas = document.getElementById('universal-camera-full');

    await ready(canvas);
</script>
