Multiple Cameras - Switcher

Create multiple Babylon.js cameras with different positions and viewing angles, then switch between them interactively.

Camera Multiple Cameras Switcher
ACTIVE CAMERA Orbit Camera
LIVE
Select a camera to change the active viewpoint.

What this demonstrates

Multiple cameras can coexist in the same scene and the active camera can be changed through Babylon.js.

Multiple Cameras

The scene contains one ArcRotateCamera and three UniversalCamera instances, each positioned at a different angle around the scene.

Camera Switcher

Select a camera from the control panel to make it the active camera. The active camera is highlighted and reported at the top of the panel.

Source

Four cameras are declared in the runtime scene and selected from JavaScript. 

<canvas id="multiple-cameras">

    <arashtad-runtime resize="engine.resize()">
    </arashtad-runtime>

    <arashtad-camera
        type="ArcRotateCamera"
        id="orbit-camera"
        args="BABYLON.Tools.ToRadians(45), BABYLON.Tools.ToRadians(65), 12, new BABYLON.Vector3(0, 1, 0)">
    </arashtad-camera>

    <arashtad-camera
        type="UniversalCamera"
        id="front-camera"
        args="new BABYLON.Vector3(0, 3, -14)"
        rotation="new BABYLON.Vector3(0, 0, 0)">
    </arashtad-camera>

    <arashtad-camera
        type="UniversalCamera"
        id="side-camera"
        args="new BABYLON.Vector3(14, 4, 0)"
        rotation="new BABYLON.Vector3(0, BABYLON.Tools.ToRadians(90), 0)">
    </arashtad-camera>

    <arashtad-camera
        type="UniversalCamera"
        id="rear-camera"
        args="new BABYLON.Vector3(0, 5, 14)"
        rotation="new BABYLON.Vector3(0, BABYLON.Tools.ToRadians(180), 0)">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="light"

    <arashtad-mesh

<script src="../../../../lib/babylon/babylon.js"></script>

<script type="module">
    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime.1.0.0.min.js';

    const canvas = document.getElementById('multiple-cameras');
    const scene = await ready(canvas);

    const buttons = document.querySelectorAll('.camera-option');
    const activeCameraName =
        document.getElementById('active-camera-name');

    function switchCamera(name) {
        const camera = scene.getCameraByName(name);

        if (!camera) {
            return;
        }

        if (scene.activeCamera && scene.activeCamera !== camera) {
            scene.activeCamera.detachControl(canvas);
        }

        scene.activeCamera = camera;
        camera.attachControl(canvas, true);

        buttons.forEach(button => {
            const active = button.dataset.camera === name;

            button.classList.toggle('active', active);

            if (active) {
                activeCameraName.textContent =
                    button.querySelector(
                        '.camera-option-info strong'
                    ).textContent;
            }
        });
    }

    buttons.forEach(button => {
        button.addEventListener('click', () => {
            switchCamera(button.dataset.camera);
        });
    });

    switchCamera('orbit-camera');
</script>
