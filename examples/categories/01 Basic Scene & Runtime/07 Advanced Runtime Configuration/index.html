Source

Runtime configuration is declared entirely through the <arashtad-runtime> element. 

<canvas id="advanced-runtime-demo">

    <arashtad-runtime
        resize="engine.resize()"
        environment="../../../assets/environments/room.hdr"
        environmentIntensity="1"
        environmentBackground="true"
        environmentBackgroundBlur="0.1"
        environmentRotationY="1"
        OnLoadProgress="document.getElementById('runtime-status').textContent = 'Loading: ' + progress + '%';">
    </arashtad-runtime>

    <arashtad-camera
        type="ArcRotateCamera"
        id="camera"
        alpha="1.57"
        beta="1.2"
        radius="8"
        target="new BABYLON.Vector3(0, 0, 0)">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="light"
        direction="new BABYLON.Vector3(0, 1, 0)"
        intensity="1.2">
    </arashtad-light>

    <arashtad-mesh
        type="Box"
        id="box"
        size="2">
    </arashtad-mesh>

</canvas>
