Source

The runtime calls the OnLoadError expression when a model fails to load. 

<canvas id="error-demo">

    <arashtad-runtime
        OnLoadError="document.getElementById('error-status').textContent = error?.message || error;">
    </arashtad-runtime>

    <arashtad-camera
        type="ArcRotateCamera"
        id="camera"
        alpha="1.57"
        beta="1.2"
        radius="8"
        target="new BABYLON.Vector3(0, 0, 0)">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="light"
        direction="new BABYLON.Vector3(0, 1, 0)"
        intensity="1.2">
    </arashtad-light>

    <arashtad-model
        id="error-model"
        src="models/missing-model.glb">
    </arashtad-model>

</canvas>

<script src="../../../../lib/babylon/babylon.js"></script>

<script type="module">
    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime.1.0.0.min.js';

    const canvas = document.getElementById('error-demo');

    await ready(canvas);
</script>
