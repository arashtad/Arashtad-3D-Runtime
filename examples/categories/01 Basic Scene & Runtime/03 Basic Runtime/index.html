Source

The runtime configuration is declared inside the canvas while JavaScript only starts the runtime. 

<canvas id="runtime-demo">

    <arashtad-runtime></arashtad-runtime>

    <arashtad-camera
        type="ArcRotateCamera"
        id="camera"
        alpha="1.57"
        beta="1.2"
        radius="8"
        target="new BABYLON.Vector3(0, 0, 0)">
    </arashtad-camera>

    <arashtad-light
        type="HemisphericLight"
        id="light"
        direction="new BABYLON.Vector3(0, 1, 0)"
        intensity="1.2">
    </arashtad-light>

    <arashtad-mesh
        type="Box"
        id="box"
        size="2">
    </arashtad-mesh>

</canvas>

<script src="../../../../lib/babylon/babylon.js"></script>

<script type="module">
    import { ready } from '../../../../lib/arashtad/arashtad-3d-runtime.1.0.0.min.js';

    const canvas = document.getElementById('runtime-demo');

    await ready(canvas);
</script>
